public boolean rateLimitCheck(Long tenantId, Long brandId, String channelType) {
    // 1. 每分钟限额
    String minSql = "SELECT COALESCE(SUM(CASE WHEN t.sent_at >= DATE_SUB(NOW(), INTERVAL 1 MINUTE) THEN 1 ELSE 0 END), 0) AS per_min " +
            "FROM marketing_touch_log t " +
            "WHERE t.tenant_id = ? AND t.channel_type = ? AND t.deleted = 0";
    Long perMin = jdbc.queryForObject(minSql, Long.class, tenantId, channelType);
    MktChannelConfig cfg = channelConfigMapper.selectOne(...tenantId+brandId+channelType+status=1 LIMIT 1);
    if (cfg != null && cfg.getRateLimitPerMin() != null && perMin != null && perMin >= cfg.getRateLimitPerMin()) return false;
    // 2. 每日限额
    String daySql = "SELECT COUNT(*) FROM marketing_touch_log t WHERE t.tenant_id = ? AND t.channel_type = ? AND t.sent_at >= DATE(NOW()) AND t.deleted = 0";
    Long perDay = jdbc.queryForObject(daySql, Long.class, tenantId, channelType);
    if (cfg != null && cfg.getRateLimitPerDay() != null && perDay != null && perDay >= cfg.getRateLimitPerDay()) return false;
    return true;
}

=== 静态分析结论 ===
1. rateLimitCheck 接受 (tenantId, brandId, channelType) 三参数
2. cfg 通过 (tenantId, brandId, channelType, status=1) 4 字段联合匹配 — status=0 配不上 → null
3. cfg=null 时不抛错，直接 return true（不限流）— 与空数据时返回 false 不一致
4. perMin/perDay 单独判断：先 perMin → 再 perDay
5. 调用方 MarketingAutomationService.startTask (line 257) — 触达每条之前调用
